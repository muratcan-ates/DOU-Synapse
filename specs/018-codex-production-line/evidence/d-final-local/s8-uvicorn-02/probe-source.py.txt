def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def candidate_hashes() -> dict[str, str]:
    return {name: digest(REPO / relative) for name, relative in SOURCES.items()}


def read_log(path: Path) -> tuple[str, list[dict]]:
    assert path.stat().st_size <= MAX_LOG_BYTES, "EXCESSIVE_PROCESS_LOG"
    raw = path.read_text()
    records = []
    for line in raw.splitlines():
        try:
            value = json.loads(line)
        except json.JSONDecodeError:
            continue
        if isinstance(value, dict):
            records.append(value)
    return raw, records


def child_environment() -> dict[str, str]:
    # Yeni cwd'de .env yoktur; ambient PG/proxy/kimlik ayarları miras alınmaz.
    return {
        "PATH": os.environ["PATH"],
        "LANG": "en_US.UTF-8",
        "PYTHONPATH": str(REPO / "apps/api"),
        "PYTHONDONTWRITEBYTECODE": "1",

def run_variant(name: str, driver: Path, old: Path) -> dict:
    logpath = OUT / f"{name}.log"
    env = child_environment()
    expected_hashes = EXPECTED_BASELINE if name == "baseline" else EXPECTED_CANDIDATE
    expected_loaded = {
        short: {
            "path": str((old / short if name == "baseline" else REPO / relative).resolve()),
            "sha256": expected_hashes[short],
        }
        for short, relative in SOURCES.items()
    }
    env["S8_EXPECTED_LOADED_SOURCES"] = json.dumps(expected_loaded)
    if name == "baseline":
        env["S8_OLD_SOURCE"] = str(old)
    # Handler boş sözlükle kurulunca parent'ın sistem/ambient proxy keşfi de kapanır.
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    responses = []
    listener = socket.socket()
    listener.bind(("127.0.0.1", 0))
    listener.listen(64)
    port = listener.getsockname()[1]
    control_read, control_write = os.pipe()
    env["S8_SOCKET_FD"] = str(listener.fileno())
    env["S8_CONTROL_FD"] = str(control_read)
    proc = None
    started = time.monotonic()
    try:
        with logpath.open("x") as stream:
            proc = subprocess.Popen(
                [str(PYTHON), str(driver)],
                cwd=OUT,
                env=env,
                pass_fds=(listener.fileno(), control_read),
                stdout=stream,
                stderr=subprocess.STDOUT,
            )
            os.close(control_read)
            control_read = -1
            deadline = time.monotonic() + 20
            while True:
                assert proc.poll() is None, "SERVER_EXITED_DURING_STARTUP"
                _, records = read_log(logpath)
                completed = any(
                    r.get("logger") == "uvicorn.error"
                    and r.get("message") == "Application startup complete."
                    for r in records
                )
                ready = [r for r in records if r.get("probe") == "server-ready"]
                if completed and ready:
                    assert ready == [
                        {"probe": "server-ready", "pid": proc.pid, "address": ["127.0.0.1", port]}
                    ], "OWNED_LISTENER_IDENTITY_MISMATCH"
                    break
                assert time.monotonic() < deadline, "SERVER_STARTUP_TIMEOUT"
                time.sleep(0.02)

            for index, (path, expected_status, template) in enumerate(REQUESTS):
                request_id = f"s8-{name}-{index}"
                request = urllib.request.Request(
                    f"http://127.0.0.1:{port}{path}", headers={"X-Request-ID": request_id}
                )
                try:
                    response = opener.open(request, timeout=5)
                except urllib.error.HTTPError as error:
                    response = error
                with response:
                    status, returned_id, body = (
                        response.status,
                        response.headers.get("X-Request-ID"),
                        response.read(),
                    )
                assert status == expected_status and returned_id == request_id, "RESPONSE_CONTRACT"
                if status == 401:
                    error = json.loads(body)["error"]
                    assert error["code"] == "unauthenticated" and error["request_id"] == request_id
                responses.append(
                    {
                        "request_id": request_id,
                        "status": status,
                        "expected_route_template": template,
                        "expected_baseline_path": path.partition("?")[0],
                        "response_sha256": hashlib.sha256(body).hexdigest(),
                    }
                )
            assert os.write(control_write, b"S") == 1
            os.close(control_write)
            control_write = -1
            proc.wait(timeout=10)
    finally:
        listener.close()
        if control_read >= 0:
            os.close(control_read)
        if control_write >= 0:
            os.close(control_write)
        if proc is not None and proc.poll() is None:
            # Yalnız sahip olunan süreç; bu hata yolu başarı sayılmaz.
            proc.kill()
            proc.wait(timeout=5)

    raw, records = read_log(logpath)
    source_binding = [r for r in records if r.get("probe") == "source-binding"]
    assert len(source_binding) == 1 and source_binding[0]["pid"] == proc.pid
    assert source_binding[0]["loaded"] == expected_loaded
    guard = [r for r in records if r.get("probe") == "database-connect-guard"]
    assert guard == [
        {
            "probe": "database-connect-guard",
            "pid": proc.pid,
            "attempts": 0,
            "control_stop_received": True,
            "control_thread_stopped": True,
        }
    ], "NORMAL_EXIT_DATABASE_GUARD_MISSING"
    assert proc.returncode == 0, "ABNORMAL_PROCESS_EXIT"
    assert not any(r.get("level") in ("ERROR", "CRITICAL") for r in records), "UNEXPECTED_SERVER_ERROR"
    access = [r for r in records if r.get("logger") == "uvicorn.access"]
    request_records = [
        r for r in records
        if r.get("logger") == "app.request" and r.get("message") == "istek tamamlandı"
    ]
    assert len(request_records) == len(REQUESTS)
    for response in responses:
        matched = [
            r for r in request_records
            if r.get("context", {}).get("request_id") == response["request_id"]
        ]
        assert len(matched) == 1
        context = matched[0]["context"]
        assert set(context) == {"request_id", "method", "path", "status", "duration_ms"}
        assert context["status"] == response["status"] and context["method"] == "GET"
        response["logged_path"] = context["path"]
        key = "expected_route_template" if name == "candidate" else "expected_baseline_path"
        assert context["path"] == response[key], "LOGGED_PATH_CONTRACT"
    presence = {item: item in raw for item in CANARIES}
    if name == "baseline":
        assert all(presence.values()) and len(access) == len(REQUESTS), "BASELINE_CONTROL_FAILED"
    else:
        assert not any(presence.values()) and not access, "CANDIDATE_PRIVACY_FAILED"
    startup_preserved = any(
        r.get("logger") == "uvicorn.error" and r.get("message") == "Application startup complete."
        for r in records
    )
    shutdown_preserved = any(
        r.get("logger") == "uvicorn.error" and r.get("message") == "Application shutdown complete."
        for r in records
    )
    assert startup_preserved and shutdown_preserved
    return {
        "variant": name,
        "seconds": round(time.monotonic() - started, 6),
        "pid": proc.pid,
        "process_exit_code": proc.returncode,
        "responses": responses,
        "raw_canary_presence": presence,
        "uvicorn_access_records": len(access),
        "uvicorn_error_startup_preserved": startup_preserved,
        "uvicorn_error_shutdown_preserved": shutdown_preserved,
        "source_binding": source_binding[0],
        "database_guard": guard[0],
        "log_sha256": digest(logpath),
    }


def main() -> int:
    os.umask(0o077)
    assert candidate_hashes() == EXPECTED_CANDIDATE, "CANDIDATE_SOURCE_MISMATCH"
    OUT.mkdir(mode=0o700, exist_ok=False)
    driver = OUT / "server.py"
    driver.write_text(DRIVER)
    old = OUT / "baseline"
    old.mkdir(mode=0o700)
    for short, relative in SOURCES.items():
        (old / short).write_bytes(
            subprocess.check_output(["git", "show", f"{BASELINE_COMMIT}:{relative}"], cwd=REPO)
        )
    assert {name: digest(old / name) for name in SOURCES} == EXPECTED_BASELINE
    record = {
        "status": "running",
        "candidate_source_hashes": EXPECTED_CANDIDATE,
        "baseline_source_hashes": EXPECTED_BASELINE,
        "baseline_commit": BASELINE_COMMIT,
        "probe_sha256": digest(Path(__file__)),
        "driver_sha256": digest(driver),
        "results": [],
        "scope": (
            "Actual Uvicorn h11 + app.main lifespan; same current repository dependencies with "
            "two pinned baseline modules vs two pinned candidate modules. Three unauthenticated "
            "requests, warmup disabled, psycopg call guard. Private pipe requests normal shutdown; "
            "no Uvicorn signal-handler override. No proxy discovery. Does not cover hosting/proxy "
            "logs, arbitrary exception content or anonymity. Historical run01 unchanged."
        ),
    }
    started = time.monotonic()
    try:
        for name in ("baseline", "candidate"):
            record["results"].append(run_variant(name, driver, old))
        assert candidate_hashes() == EXPECTED_CANDIDATE
        assert {name: digest(old / name) for name in SOURCES} == EXPECTED_BASELINE
        assert digest(driver) == record["driver_sha256"]
        record["status"] = "PASS"
        return 0
    except Exception as error:
        record["status"] = "FAIL"
        record["error_type"] = type(error).__name__
        # Assertion mesajları bu çalıştırıcıdaki sabit kontrol adlarıdır.
        record["reason"] = str(error) if isinstance(error, AssertionError) else "RUN_FAILED"
        return 2
    finally:
        record["source_unchanged"] = candidate_hashes() == EXPECTED_CANDIDATE
        record["seconds"] = round(time.monotonic() - started, 6)
        (OUT / "result.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
        print(json.dumps({"status": record["status"], "result": str(OUT / "result.json")}))


if __name__ == "__main__":
    sys.exit(main())
