def safe_files(directory):
 for p in sorted(directory.rglob('*')):
  relative=p.relative_to(directory)
  if p.is_symlink():
   excluded.append({'path':str(p),'reason':'symlink not followed'});continue
  if not p.is_file():continue
  if set(relative.parts)&{'storage','__pycache__','.ruff_cache','.pytest_cache','node_modules','.git'} or p.suffix in {'.pgpass','.pyc'} or p.name=='.env' or p.name.startswith('.env.'):
   excluded.append({'path':str(p),'reason':'storage, private credential or runtime cache excluded'});continue
  yield p
