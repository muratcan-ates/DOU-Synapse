# Overlong privacy comments are prose, not executable code. Wrap all remaining
# single comments; line-level suppressions explicitly document bounded controls.
for name in ['e2e_audit_guard.py','run_owned_e2e.py']:
    p=repo/'scripts'/name
    lines=p.read_text().splitlines()
    import textwrap
    result=[]
    for line in lines:
        if len(line)>100 and line.lstrip().startswith('#'):
            indent=line[:len(line)-len(line.lstrip())]+'# '
            result.extend(textwrap.wrap(line.lstrip()[2:],width=100,initial_indent=indent,subsequent_indent=indent))
        else:
            result.append(line)
    p.write_text('\n'.join(result)+'\n')
cp=subprocess.run([str(repo/'apps/api/.venv/bin/ruff'),'format','--no-cache','--config',str(repo/'apps/api/pyproject.toml'),*[str(repo/'scripts'/n) for n in names]],capture_output=True)
(out/'format.log').write_bytes(cp.stdout+cp.stderr)
assert cp.returncode==0
records={}
for n,data in before.items():
    after=(repo/'scripts'/n).read_bytes()
    same=ast.dump(ast.parse(data),include_attributes=False)==ast.dump(ast.parse(after),include_attributes=False)
    records[n]={'before_sha256':hashlib.sha256(data).hexdigest(),'after_sha256':hashlib.sha256(after).hexdigest(),'exact_ast_equal':same}
    assert same,n
(out/'result.json').write_text(json.dumps(records,indent=2)+'\n')
print(json.dumps(records))
